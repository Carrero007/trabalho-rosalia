CREATE DATABASE SistemaClientes;

CREATE TABLE CLIENTE (
    Id_Cliente INT IDENTITY(1,1) PRIMARY KEY,

    Tipo_Cliente VARCHAR(2) NOT NULL,  -- PF ou PJ
    Nome_RazaoSocial VARCHAR(150) NOT NULL,
    Email VARCHAR(120) NOT NULL,
    Telefone VARCHAR(20) NOT NULL,
    Endereco VARCHAR(200) NULL
);

CREATE TABLE DETALHES_PF_PJ (
    Id_Detalhes INT IDENTITY(1,1) PRIMARY KEY,
    Id_Cliente INT NOT NULL UNIQUE,  -- Relacionamento 1:1

    -- Dados PF
    CPF VARCHAR(14) NULL,
    RG VARCHAR(20) NULL,
    Data_Nascimento DATE NULL,

    -- Dados PJ
    CNPJ VARCHAR(18) NULL,
    Inscricao_Estadual VARCHAR(20) NULL,
    Nome_Responsavel VARCHAR(150) NULL,
    Data_Abertura DATE NULL,

    CONSTRAINT FK_Cliente_Detalhes
        FOREIGN KEY (Id_Cliente) REFERENCES CLIENTE(Id_Cliente)
);

CREATE UNIQUE INDEX UQ_DETALHES_CPF
ON DETALHES_PF_PJ (CPF)
WHERE CPF IS NOT NULL;

CREATE UNIQUE INDEX UQ_DETALHES_CNPJ
ON DETALHES_PF_PJ (CNPJ)
WHERE CNPJ IS NOT NULL;
